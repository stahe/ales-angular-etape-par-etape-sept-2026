    // =========================================================================
// 10-components/src/app/doctor-card/doctor-card.ts - un componente HIJO
// -------------------------------------------------------------------------
// Recibe sus datos de su padre mediante INPUTS, declarados
// con input(): cada input es un SIGNAL de solo lectura, que sigue el
// valor pasado por el padre. Su tipo se verifica en tiempo de compilación:
// <app-doctor-card [doctor]="42"> sería rechazado.
//
// Regla de oro: un componente NUNCA MODIFICA sus inputs. Los datos
// bajan del padre al hijo; para solicitar un cambio,
// el hijo emite un evento (ver ejemplo 11).
// =========================================================================

import { booleanAttribute, Component, computed, input } from '@angular/core';
import { Doctor } from '../doctor';

@Component({
  // el selector: la etiqueta que representa este componente en su padre
  selector: 'app-doctor-card',
  templateUrl: './doctor-card.html',
  styleUrl: './doctor-card.css',
})
export class DoctorCard {
  // input.required: el padre DEBE proporcionar este input
  readonly doctor = input.required<Doctor>();
  readonly rank = input.required<number>();
  // input(valor por defecto): un input opcional. booleanAttribute
  // permite escribir simplemente <app-doctor-card compact> para true
  readonly compact = input(false, { transform: booleanAttribute });

  // un computed basado en un input: se recalcula si el padre lo cambia
  protected readonly initials = computed(() => `${this.doctor().firstName[0]}${this.doctor().lastName[0]}`);
}
    • línea 19: el selector: el padre escribirá [<app-doctor-card>];
    • líneas 25–29: las entradas del componente ([inputs]), los datos que le pasa su padre. [input()] crea una señal de solo lectura que sigue el valor pasado por el padre:
      • líneas 25–26: [input.required<T>()]: una entrada obligatoria. Omitir [[doctor]] en la plantilla del padre produce un error de compilación;
      • Línea 29: [input(default value)]: una entrada opcional. La opción [transform] convierte el valor recibido: [booleanAttribute] convierte la presencia del atributo ([<app-doctor-card compact>], sin valor) en [true], como con el atributo HTML [disabled];
    • línea 32: un [computed] basado en una entrada: se recalcula si el padre cambia la entrada.

    Las entradas no son [protected]: forman parte de la interfaz pública del componente, la que utiliza su padre.

    Regla de oro: un componente nunca modifica sus entradas (que, de todos modos, son de solo lectura). Los datos fluyen del padre al hijo; para solicitar un cambio, el hijo emite un evento (ejemplo 11).

// 10-components/src/app/app.ts - el componente PADRE
// -------------------------------------------------------------------------
// Importa el componente DoctorCard (arreglo imports) y lo usa como
// una nueva etiqueta HTML, <app-doctor-card>, tantas veces como
// sea necesario: un componente es una pieza de interfaz REUTILIZABLE, con
// sus propios datos, su plantilla y su estilo.
// =========================================================================

import { Component, signal } from '@angular/core';
import { FormsModule } from '@angular/forms';
import { DoctorCard } from './doctor-card/doctor-card';
import { Doctor } from './doctor';

@Component({
  selector: 'app-root',
  imports: [FormsModule, DoctorCard],
  templateUrl: './app.html',
})
export class App {
  protected readonly doctors = signal<Doctor[]>([
    { id: 1, title: 'Mme', firstName: 'Marie', lastName: 'PELISSIER', specialty: 'general practitioner', available: true },
    { id: 2, title: 'Mr', firstName: 'Paul', lastName: 'TATOU', specialty: 'pediatrician', available: false },
    { id: 3, title: 'Melle', firstName: 'Sophie', lastName: 'DUPONT', specialty: 'general practitioner', available: true },
  ]);
  protected readonly compact = signal(false);

  // el padre modifica SUS datos: los hijos afectados se actualizan
  protected toggleAvailability(): void {
    this.doctors.update((list) => list.map((m) => ({ ...m, available: !m.available })));
  }
}

    // =========================================================================
// 11-outputs-model/src/app/counter/counter.ts
// -------------------------------------------------------------------------
// Un hijo que NOTIFICA a su padre: output() declara un evento que el
// componente puede emitir, con el tipo de sus datos. El padre lo escucha
// como un evento HTML: <app-counter (incremented)="..." (limit)="...">.
// =========================================================================

import { Component, input, output, signal } from '@angular/core';

@Component({
  selector: 'app-counter',
  template: `
    <div class="d-flex align-items-center gap-2 mb-2">
      <span class="w-25">{{ label() }}</span>
      <button class="btn btn-sm btn-outline-primary" (click)="plus()">+1</button>
      <span class="badge text-bg-secondary">{{ value() }} / {{ max() }}</span>
    </div>
  `,
})
export class Counter {
  readonly label = input.required<string>();
  readonly max = input.required<number>();

  // dos eventos: incremented (con el nuevo valor) y limit (sin datos)
  readonly incremented = output<number>();
  readonly limit = output<void>();

  protected readonly value = signal(0);

  protected plus(): void {
    if (this.value() >= this.max()) {
      this.limit.emit();
      return;
    }
    this.value.update((v) => v + 1);
    this.incremented.emit(this.value());
  }
}
    • líneas 13–19: la plantilla del componente, escrita en el decorador (propiedad [template], entre acentos graves);
    • líneas 26–27: [output<T>()] declara una salida: un evento que el componente puede emitir, junto con el tipo de sus datos. [limit] no tiene datos ([void]);
    • líneas 33 y 37: [emit(data)] emite el evento; el padre recibe los datos.

    Evite nombrar una salida con el nombre de un evento del navegador ([change], [click]...): los eventos del navegador emitidos por los elementos del componente se propagan hasta su elemento anfitrión, y el padre también los recibiría.

// 11-outputs-model/src/app/text-field/text-field.ts
// -------------------------------------------------------------------------
// Un campo de formulario "casero" (etiqueta + campo + ayuda + contador de
// caracteres) que acepta un enlace BIDIRECCIONAL, como ngModel:
//     <app-text-field [(value)]="lastName" label="Last name" [max]="20" />
//
// model() crea un signal MODIFICABLE enlazado con el padre: LEERLO da el
// valor del padre, ESCRIBIRLO (set) emite el evento valueChange, que la
// sintaxis [( )] convierte en una actualización del signal del padre. Los datos
// siguen siendo así realmente propiedad del padre.
// =========================================================================

import { Component, input, model } from '@angular/core';

// un contador compartido por todas las instancias: un identificador único por
// campo, para enlazar <label for> y <input id>
let nextNumber = 0;

@Component({
  selector: 'app-text-field',
  template: `
    <div class="mb-3">
      <label class="form-label" [for]="id">{{ label() }}</label>
      <!-- [value]: el valor del padre; (input): cada pulsación de tecla lo reemplaza -->
      <input class="form-control" [id]="id" [value]="value()" [attr.maxlength]="max()"
             [class.is-invalid]="value().length === max()" (input)="onInput($event)">
      <div class="form-text">{{ help() }} ({{ value().length }} / {{ max() }})</div>
    </div>
  `,
})
export class TextField {
  readonly value = model.required<string>();
  readonly label = input.required<string>();
  readonly max = input.required<number>();
  readonly help = input('');

  protected readonly id = `text-field-${++nextNumber}`;

  protected onInput(event: Event): void {
    this.value.set((event.target as HTMLInputElement).value);
  }
}
    • Línea 33: [model()] crea una señal modificable vinculada al padre. LEERLA devuelve el valor del padre; ESCRIBIR en ella ([set], línea 41) cambia el valor Y emite el evento [valueChange] (el nombre del modelo seguido de [Change]). Así, el padre que escribe [[(value)]="signal"] ve actualizada su señal;
    • línea 18: una variable de módulo (fuera de la clase) es compartida por todas las instancias del componente: un contador que asigna un identificador único a cada campo (línea 38), para vincular [<label for>] y [<input id>];
    • líneas 26–27: [[value]] muestra el valor del padre; [(input)] lo reemplaza con cada pulsación de tecla (línea 41). [[attr.maxlength]] es un enlace de atributo a HTML (el atributo [maxlength]), no un enlace de propiedad.

    // =========================================================================
// 11-outputs-model/src/app/app.ts - el padre escucha a sus hijos
// -------------------------------------------------------------------------
// Dos maneras de que un hijo pase información hacia arriba:
//   1. emitir un evento (output), que el padre escucha con ( );
//   2. un enlace bidireccional [( )] sobre un model(): un atajo
//      para el input [value] + el evento (valueChange).
// =========================================================================

import { Component, computed, signal } from '@angular/core';
import { Counter } from './counter/counter';
import { TextField } from './text-field/text-field';

@Component({
  selector: 'app-root',
  imports: [Counter, TextField],
  templateUrl: './app.html',
})
export class App {
  protected readonly total = signal(0);
  protected readonly message = signal('');
  protected readonly firstName = signal('Marie');
  protected readonly lastName = signal('PELISSIER');
  protected readonly fullName = computed(() => `${this.firstName()} ${this.lastName().toUpperCase()}`);

  // el manejador del evento incremented: recibe los datos emitidos ($event)
  protected onChange(label: string, value: number): void {
    this.total.update((t) => t + 1);
    this.message.set(`${label} is now ${value}`);
  }

  protected clear(): void {
    this.firstName.set('');
    this.lastName.set('');
  }
}

// 12-projection/src/app/card/card.ts - una PLANTILLA reutilizable
// -------------------------------------------------------------------------
// Los inputs pasan DATOS; la PROYECCIÓN de contenido pasa
// CONTENIDO (HTML, otros componentes), escrito por el padre ENTRE las
// etiquetas <app-card> y </app-card>. La tarjeta define el marco (borde,
// encabezado, pie), el padre proporciona lo que va dentro:
//   - <ng-content />                   : el contenido "principal";
//   - <ng-content select="[header]" /> : solo los elementos que tienen
//                                        el atributo header;
//   - el contenido colocado ENTRE <ng-content> y </ng-content> se muestra si
//     el padre no proporciona nada (contenido por defecto).
// Es el equivalente del lado cliente de la plantilla layout.hbs y su {{{body}}}
// en el curso de NestJS.
// =========================================================================

import { Component, input } from '@angular/core';

@Component({
  selector: 'app-card',
  template: `
    <div class="card mb-3" [class]="color() ? 'border-' + color() : ''">
      <div class="card-header" [class]="color() ? 'text-bg-' + color() : ''">
        <ng-content select="[header]">Untitled</ng-content>
      </div>
      <div class="card-body">
        <ng-content />
      </div>
      <!-- un pie vacío (el padre no proporcionó nada) se oculta mediante el CSS :empty -->
      <div class="card-footer text-muted small"><ng-content select="[footer]" /></div>
    </div>
  `,
  styles: `
    .card-footer:empty {
      display: none;
    }
  `,
})
export class Card {
  readonly color = input<'primary' | 'success' | 'danger'>();
}
    • línea 27: [<ng-content />]: el lugar donde [Angular] coloca el contenido escrito por el padre entre [<app-card>] y [</app-card>];
    • línea 24: [<ng-content select="[header]">]: solo los elementos de contenido que tienen el atributo [header]. [select] acepta un selector CSS (una etiqueta, una clase o un atributo). El texto entre [<ng-content>] y [</ng-content>] se muestra si el elemento padre no proporciona ningún contenido: este es el contenido predeterminado;
    • línea 30: el pie: los elementos que tienen el atributo [footer];
    • líneas 33–37: un pie vacío (el elemento padre no proporcionó nada) se oculta: la pseudoclase CSS [:empty] designa un elemento sin contenido;
    • Línea 40: una entrada opcional que solo puede tomar tres valores; si no se especifica un valor predeterminado, es [undefined] de forma predeterminada.

// 13-lifecycle/src/app/clock/clock.ts
// -------------------------------------------------------------------------
// El ciclo de vida de un componente:
//   1. constructor      : se crea la clase; los inputs aún no
//                         están disponibles;
//   2. ngOnInit         : los inputs han recibido su primer valor;
//   3. afterNextRender  : el componente se MUESTRA, sus elementos HTML
//                         existen en la página;
//   4. ngOnDestroy      : el componente está a punto de ser ELIMINADO de la página.
// Un reloj inicia un temporizador (setInterval) al arrancar... y debe
// detenerlo absolutamente cuando desaparece: de lo contrario, el temporizador
// seguiría ejecutándose en vano (fuga de memoria).
// =========================================================================

import { afterNextRender, Component, OnDestroy, OnInit, output, signal } from '@angular/core';

@Component({
  selector: 'app-clock',
  template: `<div class="display-6 font-monospace">{{ time() }}</div>`,
})
// implements OnInit, OnDestroy: TypeScript verifica que los métodos existan
export class Clock implements OnInit, OnDestroy {
  readonly log = output<string>();
  protected readonly time = signal(new Date().toLocaleTimeString());
  private timer: number | undefined;

  constructor() {
    console.log('Clock: constructor'); // el padre aún no escucha log
    // afterNextRender se registra en el constructor; la función se
    // llama una vez, justo después de la primera visualización del componente
    afterNextRender(() => this.log.emit('Clock: afterNextRender - displayed'));
  }

  ngOnInit(): void {
    this.log.emit('Clock: ngOnInit - starting the timer');
    this.timer = window.setInterval(() => this.time.set(new Date().toLocaleTimeString()), 1000);
  }

  ngOnDestroy(): void {
    window.clearInterval(this.timer);
    this.log.emit('Clock: ngOnDestroy - timer stopped');
  }
}

    Las etapas:

    • líneas 28–33: el constructor: se crea la clase. A las entradas aún no se les han asignado valores, y el padre todavía no escucha las salidas (de ahí el [console.log] de la línea 29, en lugar de un [emit]);
    • líneas 35–38: [ngOnInit()]: las entradas han recibido sus primeros valores. Este es el lugar habitual para iniciar acciones que dependen de ellas: una carga, un temporizador;
    • línea 32: [afterNextRender(function)], llamada en el constructor, registra una función que se ejecuta una sola vez, inmediatamente después de que el componente se muestre por primera vez: sus elementos HTML ya existen en la página;
    • Líneas 40–43: [ngOnDestroy()]: el componente está a punto de ser eliminado de la página. Se detiene el temporizador.

    [implements OnInit, OnDestroy] (línea 23) no es obligatorio: [Angular] llama a [ngOnInit] si existe. Pero [TypeScript] verifica entonces que los métodos estén escritos correctamente.

    3.5. Ejemplo 14: servicios e inyección de dependencias

    Un servicio es una clase que agrupa datos y código compartidos por varios componentes: el acceso al servidor, el usuario conectado, un carrito de compras, etc. Los componentes no crean servicios (nunca [new CartService()]): los solicitan a [Angular], mediante la función [inject()]. Esto es la inyección de dependencias, el mecanismo que el curso de [NestJS] presentó del lado del servidor.

    Cuando un componente solicita un servicio, [Angular] busca quién lo proporciona: el propio componente, luego su padre, el padre de su padre... hasta llegar a la aplicación. Según dónde se proporcione, un servicio se comparte en mayor o menor medida. Este ejemplo muestra tres niveles.

// 15-directives/src/app/directives/long-press.directive.ts
// -------------------------------------------------------------------------
// Una directiva también puede EMITIR eventos (output):
//     <button (appLongPress)="remove()">
// El evento solo se emite si el botón se mantiene presionado durante un segundo:

import { DestroyRef, Directive, inject, input, output, signal } from '@angular/core';

@Directive({
  selector: '[appLongPress]',
  host: {
    '[class.pressed]': 'pressing()',
    '(pointerdown)': 'start()',
    '(pointerup)': 'stop()',
    '(pointerleave)': 'stop()',
  },
})
export class LongPressDirective {
  readonly duration = input(1000); // en milisegundos
  readonly appLongPress = output<void>();
  protected readonly pressing = signal(false);
  private timer: number | undefined;

  constructor() {
    inject(DestroyRef).onDestroy(() => window.clearTimeout(this.timer));
  }

  protected start(): void {
    this.pressing.set(true);
    this.timer = window.setTimeout(() => {
      this.pressing.set(false);
      this.appLongPress.emit();
    }, this.duration());
  }

  protected stop(): void {
    window.clearTimeout(this.timer);
    this.pressing.set(false);
  }
}

    // =========================================================================
// 16-confirmation/src/app/confirmation/confirmation.service.ts
// -------------------------------------------------------------------------
// "¿Está seguro?": window.confirm() existe, pero no se le puede dar
// estilo ni traducir sus botones. Lo reemplazamos por un diálogo de
// Bootstrap (confirm-dialog.ts), controlado por este servicio.
//
// El truco: confirm(message) devuelve una PROMESA, resuelta a true (Sí)
// o false (No) cuando el usuario responde. El código que llama se escribe
// entonces tan simple como con window.confirm:
//
//     if (await this.confirmation.confirm('Delete this doctor?')) { ... }
//
// providedIn: 'root': un solo servicio para toda la aplicación, por lo tanto un solo
// diálogo, colocado una vez en el componente raíz.
// =========================================================================

import { Injectable, signal } from '@angular/core';

@Injectable({ providedIn: 'root' })
export class ConfirmationService {
  private readonly _message = signal<string | null>(null); // null: diálogo cerrado
  readonly message = this._message.asReadonly();
  private answer: ((yes: boolean) => void) | null = null;

  confirm(text: string): Promise<boolean> {
    this.answer?.(false); // ¿una pregunta aún abierta? cuenta como "no"
    this._message.set(text);
    return new Promise<boolean>((resolve) => {
      this.answer = resolve;
    });
  }

  respond(yes: boolean): void {
    this._message.set(null);
    const r = this.answer;
    this.answer = null;
    r?.(yes);
  }
}
    • línea 20: un único servicio para toda la aplicación; por lo tanto, habrá una sola ventana, colocada de una vez por todas en el componente raíz;
    • línea 22: el mensaje mostrado; [null] significa que la ventana está cerrada;
    • línea 23: [asReadonly()]: los componentes pueden leer el mensaje, pero solo el servicio puede modificarlo;
    • línea 24: la función que resolverá la promesa actual, que está a la espera de una respuesta;
    • líneas 26–32: [confirm] abre la ventana (línea 28) y devuelve una promesa. Su función [resolve] se guarda aparte (línea 30): la promesa no se resolverá hasta más tarde, cuando el usuario haga clic en un botón;
    • línea 27: si todavía hubiera una pregunta abierta, se respondería “no”: su promesa no permanecerá pendiente para siempre;
    • líneas 34–39: [respond] es llamada por la ventana: cierra la ventana (línea 35) y luego resuelve la promesa con la respuesta del usuario (línea 38).

    Esta es la ventaja de la promesa: el código que hace la pregunta la escribe como una llamada a [window.confirm()], esperando la respuesta con [await], sin función de callback.

import { Component, inject, signal } from '@angular/core';

  id: number;
  name: string;
}

  selector: 'app-root',
  imports: [ConfirmDialog],
  templateUrl: './app.html',
})

  private readonly confirmation = inject(ConfirmationService);
  protected readonly contacts = signal<Contact[]>([

    { id: 2, name: 'Alice DURAND' },
    { id: 3, name: 'Luc GERMAN' },

    if (!(await this.confirmation.confirm(`Delete the contact ${c.name}?`))) {
      this.message.set('Deletion cancelled.');
      return;

    this.contacts.update((list) => list.filter((x) => x.id !== c.id));
    this.message.set(`${c.name} has been deleted.`);

    if (await this.confirmation.confirm(`Delete the ${this.contacts().length} contacts?`)) {
      this.contacts.set([]);
      this.message.set('All the contacts have been deleted.');
    }
  }
}
