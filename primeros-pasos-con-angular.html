    // =========================================================================
// 01-first-application/src/app/app.ts - un primer componente
// -------------------------------------------------------------------------
// Un componente Angular es una CLASE de TypeScript, decorada con @Component:
//   - selector    : la etiqueta HTML que lo representa (<app-root>);
//   - templateUrl : el archivo de su HTML (app.html);
//   - styleUrl    : el archivo de su CSS (app.css), que solo se aplica
//                   a este componente.
// Las propiedades y los métodos de la clase se pueden usar en la plantilla.
// =========================================================================

import { Component, signal } from '@angular/core';

@Component({
  selector: 'app-root',
  templateUrl: './app.html',
  styleUrl: './app.css',
})
export class App {
  // datos "ordinarios": la plantilla puede mostrarlos
  protected readonly title = 'Welcome to Angular';
  protected readonly languages = ['TypeScript', 'HTML', 'CSS'];

  // signal(...): un dato REACTIVO. Su valor se lee LLAMÁNDOLO
  // (counter()) y se cambia con set() o update(). Angular actualiza entonces
  // AUTOMÁTICAMENTE los lugares de la página que lo muestran.
  protected readonly counter = signal(0);
  protected readonly time = signal(new Date().toLocaleTimeString());

  // un método llamado por el botón de la plantilla
  protected increment(): void {
    this.counter.update((n) => n + 1);
    this.time.set(new Date().toLocaleTimeString());
  }
}
    • Líneas 14–18: el decorador [@Component] convierte la clase en un componente:
      • línea 15: su selector, la etiqueta que lo representa en una plantilla o en [index.html];
      • línea 16: su archivo de plantilla, HTML;
      • línea 17: su archivo de estilo, CSS;
    • líneas 21–22: dos datos ordinarios. [protected]: solo los usan el componente y su plantilla; [readonly]: no serán reemplazados;
    • línea 27: [signal(0)] crea un dato reactivo con un valor inicial de 0. Una señal es una función: su valor se lee llamándola ([this.counter()]) y se cambia mediante [set(value)] o [update(function)]. Cuando cambia, [Angular] actualiza todos los lugares de la página que la muestran;
    • líneas 31–34: un método que modifica dos señales. No altera la página: [Angular] la actualizará. [update((n) => n + 1)] calcula el nuevo valor a partir del anterior.

    // =========================================================================
// 02-signals/src/app/app.ts - signal, computed, effect, linkedSignal
// -------------------------------------------------------------------------
// Una pequeña calculadora de precios con IVA incluido:
//   - signal       : un valor reactivo, que se lee llamándolo y
//                    se modifica con set() / update();
//   - computed     : un valor CALCULADO a partir de otros signals,
//                    recalculado automáticamente cuando estos cambian;
//   - effect       : código ejecutado en cada cambio de los signals que
//                    lee (un "efecto secundario");
//   - linkedSignal : un signal modificable, pero REINICIADO a un valor calculado
//                    cuando cambia otro signal.
// =========================================================================

import { Component, computed, effect, linkedSignal, signal, untracked } from '@angular/core';
// FormsModule aporta [(ngModel)], que enlaza un campo con un signal (ver ejemplo 05)
import { FormsModule } from '@angular/forms';

@Component({
  selector: 'app-root',
  imports: [FormsModule],
  templateUrl: './app.html',
})
export class App {
  protected readonly label = signal('Pen');
  protected readonly priceExclVat = signal(2.5);
  protected readonly vatRate = signal(20); // en %

  // linkedSignal: la cantidad vuelve a 1 cada vez que cambia el artículo,
  // pero el usuario puede modificarla mientras tanto
  protected readonly quantity = linkedSignal({ source: this.label, computation: () => 1 });

  // computed: nunca se asigna directamente, siempre se recalcula. Angular sabe
  // de qué signals depende (los que LEE): lo recalcula cuando uno
  // de ellos cambia, y solo en ese caso
  protected readonly totalExclVat = computed(() => this.quantity() * this.priceExclVat());
  protected readonly vatAmount = computed(() => (this.totalExclVat() * this.vatRate()) / 100);
  protected readonly totalInclVat = computed(() => this.totalExclVat() + this.vatAmount());

  // un formateador de montos en euros, estilo inglés (€1,234.50)
  private readonly euros = new Intl.NumberFormat('en-US', { style: 'currency', currency: 'EUR' });

  // el historial de los totales con IVA incluido, llenado por el effect de abajo
  protected readonly history = signal<string[]>([]);
  private oldTotal: number | null = null;

  constructor() {
    // effect: se ejecuta una primera vez y luego en cada cambio de
    // totalInclVat (el único signal que lee)
    effect(() => {
      const total = this.totalInclVat();
      document.title = `Incl. VAT: ${this.format(total)}`;
      if (this.oldTotal !== null) {
        const line = `${this.format(this.oldTotal)} -> ${this.format(total)}`;
        // untracked: modificar history SIN convertirlo en una dependencia del effect
        untracked(() => this.history.update((h) => [line, ...h].slice(0, 5)));
      }
      this.oldTotal = total;
    });
  }

  protected format(amount: number): string {
    return this.euros.format(amount);
  }

  protected reset(): void {
    this.label.set('Pen');
    this.priceExclVat.set(2.5);
    this.vatRate.set(20);
    this.quantity.set(1);
  }
}
    • línea 15: las funciones de reactividad, todas proporcionadas por [@angular/core];
    • Líneas 21 y 17: la plantilla usa [[(ngModel)]], que proviene del módulo [FormsModule]. Un componente debe importar (arreglo [imports]) todo lo que su plantilla utiliza: otros componentes, directivas y pipes. El compilador reporta un error si se omite;
    • líneas 25–27: tres señales, [signal(initial value)]. Su tipo se infiere del valor inicial: [signal('Pen')] es una señal de cadena, [signal(2.5)] es una señal numérica;
    • línea 31: una [linkedSignal] es una señal modificable (se puede escribir en ella, igual que en una señal ordinaria), pero está vinculada a otra señal, su origen: cada vez que el origen cambia (en este caso, la descripción del artículo), su valor se restablece al valor calculado por [computation] (en este caso, 1). Cuando el artículo cambia, la cantidad vuelve a 1;

import { Component, signal } from '@angular/core';

interface Task {
  id: number;
  label: string;
  done: boolean;
}

@Component({
  selector: 'app-root',
  templateUrl: './app.html',
  styles: `
    .area {
      background: #e9f2ff;
      height: 80px;
      cursor: crosshair;
    }
  `,
})
export class App {
  protected readonly tasks = signal<Task[]>([
    { id: 1, label: 'Install Node.js', done: true },
    { id: 2, label: 'Read the chapter on events', done: false },
  ]);
  protected readonly log = signal<string[]>([]);
  private nextId = 3;

  private trace(message: string): void {
    this.log.update((j) => [`${new Date().toLocaleTimeString()} ${message}`, ...j]);
  }

  // llamado por (submit): $event.preventDefault() evita que el navegador
  // envíe el formulario al servidor (y que recargue la página)
  protected add(event: SubmitEvent, field: HTMLInputElement): void {
    event.preventDefault();
    const label = field.value.trim();
    if (label === '') {
      return;
    }
    // un signal que contiene un arreglo: lo REEMPLAZAMOS por un nuevo arreglo
    this.tasks.update((t) => [...t, { id: this.nextId++, label, done: false }]);
    this.trace(`added "${label}"`);
    field.value = '';
    field.focus();
  }

  protected toggle(task: Task, done: boolean): void {
    this.tasks.update((t) => t.map((x) => (x.id === task.id ? { ...x, done } : x)));
    this.trace(`"${task.label}": ${done ? 'done' : 'to do'}`);
  }

  protected remove(task: Task): void {
    this.tasks.update((t) => t.filter((x) => x.id !== task.id));
    this.trace(`deleted "${task.label}"`);
  }

  // $event: el evento nativo (aquí un MouseEvent), con sus coordenadas
  protected position(event: MouseEvent): void {
    this.trace(`click at (${event.offsetX}, ${event.offsetY})`);
  }
}

// 06-signal-forms/src/app/app.ts - formularios con signals
// -------------------------------------------------------------------------
// La forma moderna (Angular 22) de escribir un formulario:
//   1. el MODELO: un signal que contiene un objeto, con una propiedad por
//      campo (el formulario lee y escribe en este signal);
//   2. form(model, schema): construye un ÁRBOL DE CAMPOS (FieldTree), que
//      tiene la misma forma que el modelo: f.lastName, f.age, f.newsletter... El
//      schema (opcional) describe las reglas de los campos (ver ejemplo 07);
//   3. [formField]="f.lastName" enlaza un campo HTML con un campo del árbol.
// Cada campo tiene un ESTADO, que se obtiene llamándolo: f.lastName() da su valor
// (value()), si ha sido tocado (touched()), modificado (dirty()), si está
// deshabilitado (disabled())...
// =========================================================================

import { Component, signal } from '@angular/core';
import { JsonPipe } from '@angular/common';
import { disabled, form, FormField } from '@angular/forms/signals';

// la forma de los datos del formulario
interface Profile {
  title: string;
  firstName: string;
  lastName: string;
  age: number;
  newsletter: boolean;
  email: string;
  country: string;
  birthDate: string;
  presentation: string;
}

const INITIAL: Profile = {
  title: 'Mme',
  firstName: 'Marie',
  lastName: 'PELISSIER',
  age: 30,
  newsletter: false,
  email: '',
  country: 'fr',
  birthDate: '',
  presentation: '',
};

@Component({
  selector: 'app-root',
  imports: [FormField, JsonPipe],
  templateUrl: './app.html',
})
export class App {
  protected readonly titles = ['Mr', 'Mme', 'Melle'];

  // 1. el modelo
  protected readonly model = signal<Profile>({ ...INITIAL });

  // 2. el árbol de campos; una regla: el correo electrónico solo se puede modificar si la
  // casilla "newsletter" está marcada (valueOf lee el valor de OTRO campo)
  protected readonly f = form(this.model, (p) => {
    disabled(p.email, ({ valueOf }) => !valueOf(p.newsletter));
  });

  // los campos cuyo estado se muestra, en la tabla de la derecha
  protected readonly fields = ['firstName', 'lastName', 'age', 'email'] as const;

  protected reset(): void {
    // el modelo vuelve a su valor inicial, y el estado de los campos a cero
    this.model.set({ ...INITIAL });
    this.f().reset();
  }

  protected upperCase(): void {
    // modificar el MODELO por código: los campos lo siguen
    this.model.update((m) => ({ ...m, lastName: m.lastName.toUpperCase(), firstName: m.firstName.toUpperCase() }));
  }
}

import { Component, signal } from '@angular/core';
import { form, FormField, maxLength, minLength, pattern, required, submit, validate } from '@angular/forms/signals';

interface Account {
  title: string;
  firstName: string;
  lastName: string;
  login: string;
  password: string;
  confirmation: string;
}

const TITLES = ['Mr', 'Mme', 'Melle'];
const LOGIN_FORMAT = /^[a-z0-9._-]{1,30}$/;
const PASSWORD_MIN = 6;

@Component({
  selector: 'app-root',
  imports: [FormField],
  templateUrl: './app.html',
})
export class App {

  protected readonly model = signal<Account>({
    title: 'Mr',
    firstName: '',
    lastName: '',
    login: '',
    password: '',
    confirmation: '',
  });

  // el schema: p designa las RUTAS de los campos (p.lastName, p.login...)
  protected readonly f = form(this.model, (p) => {
    validate(p.title, ({ value }) =>
      TITLES.includes(value()) ? null : { kind: 'title', message: 'Choose a title from the list.' },
    );
    required(p.firstName, { message: 'The first name is required.' });
    maxLength(p.firstName, 20, { message: 'The first name must not exceed 20 characters.' });
    required(p.lastName, { message: 'The last name is required.' });
    maxLength(p.lastName, 20, { message: 'The last name must not exceed 20 characters.' });
    required(p.login, { message: 'The login is required.' });
    pattern(p.login, LOGIN_FORMAT, { message: 'Unaccented lower-case letters, digits, dot, hyphen, underscore.' });
    required(p.password, { message: 'The password is required.' });
    minLength(p.password, PASSWORD_MIN, { message: `At least ${PASSWORD_MIN} characters.` });
    // una regla "casera", que compara DOS campos (valueOf lee el otro)
    validate(p.confirmation, ({ value, valueOf }) =>
      value() === valueOf(p.password)
        ? null
        : { kind: 'confirmation', message: 'The confirmation does not match the password.' },
    );
  });

  protected readonly submitAttempted = signal(false);
  protected readonly success = signal(false);

  // (submit) del formulario
  protected async send(event: SubmitEvent): Promise<void> {
    event.preventDefault();
    this.submitAttempted.set(true);
    // submit() marca todos los campos como touched (aparecen sus errores),
    // y luego solo ejecuta la acción si todo es válido

      // aquí enviaríamos el modelo al servidor (ver ejemplo 25)
      return null; // null: sin error

import { Component, inject, signal } from '@angular/core';
import { AbstractControl, FormBuilder, ReactiveFormsModule, ValidationErrors, Validators } from '@angular/forms';

const TITLES = ['Mr', 'Mme', 'Melle'];

// un validador "casero" se aplica al GRUPO: compara dos de sus campos.
// Devuelve null (sin error) o un objeto que describe el error.
function sameConfirmation(group: AbstractControl): ValidationErrors | null {
  const password = group.get('password')?.value;
  const confirmation = group.get('confirmation')?.value;
  return password === confirmation ? null : { confirmation: true };
}

@Component({
  selector: 'app-root',
  imports: [ReactiveFormsModule],
  templateUrl: './app.html',
})
export class App {
  protected readonly titles = TITLES;
  private readonly fb = inject(FormBuilder).nonNullable;

  // el grupo: { name: [valor inicial, validadores] }
  protected readonly form = this.fb.group(
    {
      title: ['Mr', [Validators.required]],
      firstName: ['', [Validators.required, Validators.maxLength(20)]],
      lastName: ['', [Validators.required, Validators.maxLength(20)]],
      login: ['', [Validators.required, Validators.pattern(/^[a-z0-9._-]{1,30}$/)]],
      password: ['', [Validators.required, Validators.minLength(6)]],
      confirmation: [''],
    },
    { validators: sameConfirmation },
  );

  protected readonly success = signal(false);
  protected readonly value = signal('');

  // un campo se señala con error si ha sido tocado y es inválido
  protected inError(name: string): boolean {
    const control = this.form.get(name);
    return !!control && control.touched && control.invalid;
  }

  // (ngSubmit): todos los campos pasan a "touched", aparecen sus errores
  protected send(): void {
    this.form.markAllAsTouched();
    this.success.set(this.form.valid);
    // getRawValue(): el valor de todo el grupo, tipado
    this.value.set(JSON.stringify(this.form.getRawValue()));
  }
}

    // =========================================================================
// 09-pipes/src/app/pipes/time-range.pipe.ts - un pipe "casero"
// -------------------------------------------------------------------------
// Un pipe es una clase decorada con @Pipe, cuyo método transform()
// recibe el valor colocado ANTES de la barra (y, eventualmente,
// parámetros) y devuelve lo que debe mostrarse:
//     {{ slot | timeRange }}           -> "8:00 - 8:20"
//     {{ slot | timeRange: 'start' }}  -> "8:00"
// Un pipe es "puro" por defecto: Angular solo vuelve a llamar a transform() si el
// valor o los parámetros han CAMBIADO (mismo objeto = mismo resultado).
// =========================================================================

import { Pipe, PipeTransform } from '@angular/core';

export interface Slot {
  startHour: number;
  startMinute: number;
  endHour: number;
  endMinute: number;
}

// "8:05" en lugar de "8:5"
function time(h: number, m: number): string {
  return `${h}:${String(m).padStart(2, '0')}`;
}

@Pipe({ name: 'timeRange' })
export class TimeRangePipe implements PipeTransform {
  transform(c: Slot, part: 'all' | 'start' | 'end' = 'all'): string {
    switch (part) {
      case 'start':
        return time(c.startHour, c.startMinute);
      case 'end':
        return time(c.endHour, c.endMinute);
      default:
        return `${time(c.startHour, c.startMinute)} - ${time(c.endHour, c.endMinute)}`;
    }
  }
}
    • línea 27: el nombre del pipe, el que se escribe en las plantillas;
    • línea 29: [transform(value, part)] devuelve lo que debe mostrarse. El parámetro es opcional (valor predeterminado ['all']): [{{ slot | timeRange }}] muestra [8:00 - 8:20], [{{ slot | timeRange: 'start' }}] muestra [8:00].

    Un pipe es puro de forma predeterminada: [Angular] solo llama a [transform] si el valor o alguno de los parámetros ha cambiado (un objeto distinto, una cadena distinta). Esto es lo que lo hace más eficiente que un método del componente llamado dentro de la plantilla (como [format(...)] en el ejemplo 02), que se ejecuta cada vez que se verifica el componente.

export class InitialsPipe implements PipeTransform {
